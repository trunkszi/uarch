# Baseline is compiled explicitly so ambient host ISA flags cannot decide the expected result.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
  target_compile_options(compiled_test PRIVATE -march=x86-64)
  target_compile_definitions(compiled_test PRIVATE UARCH_EXPECTED_TIER=0)
  foreach(tier IN ITEMS v3 v4)
    add_library(compiled_${tier}_test OBJECT tiers/compiled_test.cc)
    uarch_configure_internal(compiled_${tier}_test)
    target_compile_options(compiled_${tier}_test PRIVATE -march=x86-64-${tier})
    if(tier STREQUAL "v3")
      target_compile_definitions(compiled_${tier}_test PRIVATE UARCH_EXPECTED_TIER=1)
    else()
      target_compile_definitions(compiled_${tier}_test PRIVATE UARCH_EXPECTED_TIER=3)
    endif()
    # Compile target metadata without executing advanced instructions on an unaudited host.
  endforeach()
  foreach(feature IN ITEMS avx2 avx512f)
    add_library(compiled_partial_${feature}_test OBJECT tiers/compiled_test.cc)
    uarch_configure_internal(compiled_partial_${feature}_test)
    target_compile_options(compiled_partial_${feature}_test PRIVATE -march=x86-64 -m${feature})
    target_compile_definitions(compiled_partial_${feature}_test PRIVATE UARCH_EXPECTED_TIER=0)
  endforeach()
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  target_compile_definitions(compiled_test PRIVATE UARCH_EXPECTED_TIER=4)
else()
  target_compile_definitions(compiled_test PRIVATE UARCH_EXPECTED_TIER=0)
endif()
